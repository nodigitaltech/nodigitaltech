from nodigital.ai import EnterpriseAgent, KnowledgeRAG
from nodigital.infra import ProxmoxCluster, OPNSenseAPI

async def process_business_event(ticket):
  rag = await KnowledgeRAG.load_policy("LGPD_SLA")
  diagnosis = await rag.analyze(ticket.data)
  if diagnosis.requires_remediation:
    await ProxmoxCluster.auto_snapshot(ticket.vm_id)
    await OPNSenseAPI.isolate_host(ticket.ip)
  return {"status": "RESOLVED", "sla": "< 5m"}
